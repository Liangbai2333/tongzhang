```sh
adb -s emulator-5588 reverse tcp:3374 tcp:3374
adb -s emulator-5588 reverse tcp:8074 tcp:8074
adb -s emulator-5588 reverse --list
adb -s emulator-5588 shell am start -a android.intent.action.VIEW \
  -d 'tongzhang://expo-development-client/?url=http%3A%2F%2Flocalhost%3A8074' com.tongzhang.app
```
